def find(w):
    name=C.c_char_p()
    x.XFetchName(d,w,C.byref(name))
    title=name.value.decode(errors='replace') if name.value else ''
    if name: x.XFree(name)
    if title.startswith(__import__('os').environ.get('LUMEN_TITLE','LUMEN LINK')): return w
    rt=W();parent=W();children=C.POINTER(W)();count=C.c_uint()
    x.XQueryTree(d,w,C.byref(rt),C.byref(parent),C.byref(children),C.byref(count))
    ids=[children[i] for i in range(count.value)]
    if children: x.XFree(children)
    for child in ids:
        result=find(child)
        if result: return result
